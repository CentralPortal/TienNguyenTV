const asyncStorage = {
  getItem: async (key) => {
    try {
      if (window.storage && typeof window.storage.get === "function") {
        const res = await window.storage.get(key, false);
        return res ? res.value : null;
      }
      return localStorage.getItem(key);
    } catch (e) {
      return localStorage.getItem(key);
    }
  },
  setItem: async (key, value) => {
    try {
      if (window.storage && typeof window.storage.set === "function") {
        await window.storage.set(key, value, false);
        return true;
      }
      localStorage.setItem(key, value);
      return true;
    } catch (e) {
      try {
        localStorage.setItem(key, value);
        return true;
      } catch (err) {
        return false;
      }
    }
  },
};

const normalizeItems = (items) => {
  const RENAMES = [
    ["Chuồng Gỗ (8ft x 10ft) + Công Thợ", "Chuồng tầm 2.4m x 3m + Công Thợ", 13000000],
    ["Lưới Rào Khu Vực Chạy (200 sq ft)", "Lưới Rào Khu Vực Chạy tầm 18.5m²", null],
    ["Thanh Đậu Dài 20 Feet (VD: 4 thanh 5 feet)", "Thanh đậu dài 6m (VD: 4 thanh, mỗi thanh dài 1.5m)", null],
    ["20 Con Gà Giống", "25+ Con Gà Giống", null],
  ];
  const COST_FIXES = {
    "Phí Nhân Công": 1500000,
    "Chi Phí Di Chuyển (xăng, phí đường, v.v.)": 500000,
  };

  let list = (items || []).map((it) => {
    let { name, cost } = it;
    const rename = RENAMES.find(([oldName]) => oldName === name);
    if (rename) {
      name = rename[1];
      if (rename[2] !== null) cost = rename[2];
    } else if (COST_FIXES[name] !== undefined && (cost === 0 || cost === null)) {
      cost = COST_FIXES[name];
    }
    return { ...it, name, cost: Number(cost) || 0 };
  });

  if (!list.some((it) => it.name === "Nệm")) {
    list.push({ id: uid(), name: "Nệm", cost: 600000, checked: false });
  }

  return list;
};

const migrateRecord = (r) => {
  if (Array.isArray(r.items)) {
    const cleaned = r.items.filter((it) => (it.name || "").trim() !== "Xe Đạp");
    return { ...r, items: normalizeItems(cleaned) };
  }

  if (Array.isArray(r.chickenItems) || Array.isArray(r.otherItems)) {
    const merged = [
      ...(Array.isArray(r.chickenItems) ? r.chickenItems : []),
      ...(Array.isArray(r.otherItems) ? r.otherItems : []),
    ].filter((it) => (it.name || "").trim() !== "Xe Đạp");
    return { ...r, items: normalizeItems(merged.length ? merged : seedItems()) };
  }

  return { ...r, items: seedItems() };
};

function ToastNotification({ toast, onClose }) {
  if (!toast) return null;

  const bgStyles = {
    success: "bg-emerald-900 text-emerald-100 border-emerald-700 shadow-emerald-900/10",
    error: "bg-rose-900 text-rose-100 border-rose-700 shadow-rose-900/10",
    info: "bg-stone-900 text-stone-100 border-stone-800 shadow-stone-900/10",
  }[toast.type || "info"];

  return (
    <div className="fixed bottom-5 right-5 z-50 max-w-sm animate-bounce-short">
      <div className={`flex items-center gap-3 px-4 py-3 rounded-xl border shadow-xl ${bgStyles}`}>
        {toast.type === "success" && <CheckCircle2 className="w-5 h-5 text-emerald-400 shrink-0" />}
        {toast.type === "error" && <AlertTriangle className="w-5 h-5 text-rose-400 shrink-0" />}
        {toast.type === "info" && <Sparkles className="w-5 h-5 text-amber-400 shrink-0" />}
        <span className="text-sm font-medium">{toast.message}</span>
        <button
          onClick={onClose}
          className="ml-auto text-stone-400 hover:text-white transition-colors"
          aria-label="Đóng"
        >
          <X className="w-4 h-4" />
        </button>
      </div>
    </div>
  );
}

        setUser(userCred.user);

        const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'appState', 'current');

        unsubscribeDoc = onSnapshot(
          docRef,
          (snapshot) => {
            if (snapshot.exists()) {
              const data = snapshot.data();
              if (data) {
                isRemoteUpdating.current = true;
                if (data.month !== undefined) setMonth(data.month);
                if (Array.isArray(data.records)) {
                  const migrated = data.records.map(migrateRecord);
                  setRecords(migrated.length ? migrated : [makeRecord("Gia đình 1"), makeRecord("Gia đình 2")]);
                }
                setTimeout(() => {
                  isRemoteUpdating.current = false;
                }, 100);
              }
            } else {
              const initialRecs = [makeRecord("Gia đình 1"), makeRecord("Gia đình 2")];
              setRecords(initialRecs);
              setDoc(docRef, { month, records: initialRecs, updatedAt: Date.now() });
            }
            setSaveStatus("saved");
            setLoading(false);
          },
          (error) => {
            console.error("Cloud sync error:", error);
            setSaveStatus("offline");
            setLoading(false);
          }
        );
      } catch (err) {
        console.error("Auth / Cloud init error:", err);
        setSaveStatus("offline");
        setLoading(false);
      }
    };

    initAuthAndSync();

    return () => {
      isMounted = false;
      if (unsubscribeDoc) unsubscribeDoc();
    };
  }, []);

  useEffect(() => {
    if (loading || isRemoteUpdating.current) return;

    setSaveStatus("saving");
    if (saveTimer.current) clearTimeout(saveTimer.current);

    saveTimer.current = setTimeout(async () => {
      try {
        if (db && auth?.currentUser) {
          const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'appState', 'current');
          await setDoc(docRef, {
            month,
            records,
            updatedAt: Date.now(),
            updatedBy: auth.currentUser.uid
          }, { merge: true });
          setSaveStatus("saved");
        } else {
          await asyncStorage.setItem(STORAGE_KEY, JSON.stringify({ month, records }));
          setSaveStatus("saved");
        }
      } catch (e) {
        console.error("Save error:", e);
        setSaveStatus("offline");
      }
    }, 600);

    return () => clearTimeout(saveTimer.current);
  }, [month, records, loading]);

  const grandTotal = useMemo(() => {
    return records.reduce((sum, r) => sum + recordTotal(r), 0);
  }, [records]);

  const handleBackup = () => {
    try {
      const dataStr = JSON.stringify({ month, records, exportedAt: new Date().toISOString() }, null, 2);
      const blob = new Blob([dataStr], { type: "application/json" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      const dateStr = new Date().toISOString().slice(0, 10);
      a.href = url;
      a.download = `sao-luu-du-lieu-quyen-gop-${dateStr}.json`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
      showToast("Đã tải xuống tệp sao lưu dữ liệu!", "success");
    } catch (err) {
      showToast("Không thể tạo tệp sao lưu.", "error");
    }
  };

  const handleRestoreClick = () => {
    if (fileInputRef.current) fileInputRef.current.click();
  };

  const handleFileSelected = (e) => {
    const file = e.target.files && e.target.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (evt) => {
      try {
        const parsed = JSON.parse(evt.target.result);
        if (!parsed || !Array.isArray(parsed.records)) {
          showToast("Tệp sao lưu không hợp lệ.", "error");
          return;
        }
        setModal({
          open: true,
          type: "restore",
          recordId: null,
          itemId: null,
          payload: parsed,
        });
      } catch (err) {
        showToast("Lỗi định dạng tệp JSON.", "error");
      }
    };
    reader.readAsText(file);
    e.target.value = "";
  };

  const updateField = (recordId, field, value) => {
    setRecords((prev) => prev.map((r) => (r.id === recordId ? { ...r, [field]: value } : r)));
  };

  const toggleExpand = (recordId) => {
    setRecords((prev) =>
      prev.map((r) => (r.id === recordId ? { ...r, expanded: !r.expanded } : r))
    );
  };

  const toggleItem = (recordId, itemId, checked) => {
    setRecords((prev) =>
      prev.map((r) =>
        r.id === recordId
          ? {
              ...r,
              items: r.items.map((it) => (it.id === itemId ? { ...it, checked } : it)),
            }
          : r
      )
    );
  };

  const changeItemName = (recordId, itemId, name) => {
    setRecords((prev) =>
      prev.map((r) =>
        r.id === recordId
          ? {
              ...r,
              items: r.items.map((it) => (it.id === itemId ? { ...it, name } : it)),
            }
          : r
      )
    );
  };

  const changeItemCost = (recordId, itemId, cost) => {
    setRecords((prev) =>
      prev.map((r) =>
        r.id === recordId
          ? {
              ...r,
              items: r.items.map((it) => (it.id === itemId ? { ...it, cost } : it)),
            }
          : r
      )
    );
  };

  const addItem = (recordId) => {
    setRecords((prev) =>
      prev.map((r) =>
        r.id === recordId
          ? {
              ...r,
              items: [...r.items, { id: uid(), name: "", cost: 0, checked: true }],
            }
          : r
      )
    );
  };

  const addRecord = () => {
    const nextNum = records.length + 1;
    const newRec = makeRecord(`Gia đình ${nextNum}`);
    setRecords((prev) => [...prev, newRec]);
    showToast(`Đã thêm Gia đình ${nextNum}`, "success");
    setTimeout(() => {
      const el = document.getElementById(`record-${newRec.id}`);
      if (el) el.scrollIntoView({ behavior: "smooth", block: "start" });
    }, 100);
  };

  const requestDelete = (recordId) => {
    setModal({ open: true, type: "delete", recordId, itemId: null });
  };

  const requestDeleteItem = (recordId, itemId) => {
    setModal({ open: true, type: "deleteItem", recordId, itemId });
  };

  const closeModal = () =>
    setModal({ open: false, type: null, recordId: null, itemId: null });

  const confirmModal = () => {
    if (modal.type === "delete") {
      setRecords((prev) => prev.filter((r) => r.id !== modal.recordId));
      showToast("Đã xóa gia đình thành công", "info");
    } else if (modal.type === "deleteItem") {
      setRecords((prev) =>
        prev.map((r) =>
          r.id === modal.recordId
            ? { ...r, items: r.items.filter((it) => it.id !== modal.itemId) }
            : r
        )
      );
      showToast("Đã xóa vật phẩm", "info");
    } else if (modal.type === "restore" && modal.payload) {
      const recs = Array.isArray(modal.payload.records) ? modal.payload.records : [];
      const migrated = recs.map(migrateRecord);
      setMonth(modal.payload.month || month);
      setRecords(migrated.length ? migrated : records);
      showToast("Khôi phục dữ liệu thành công! Đang đồng bộ...", "success");
    }
    closeModal();
  };

  if (loading) {
    return (
      <div className="min-h-screen bg-stone-50 text-stone-600 flex items-center justify-center font-mono text-sm gap-3">
        <RefreshCw className="w-5 h-5 animate-spin text-amber-600" />
        Đang tải dữ liệu hỗ trợ...
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-stone-100/70 text-stone-800 font-sans pb-16 selection:bg-amber-200">
      {/* Top Header & Dashboard Total Display */}
      <header className="bg-white border-b border-stone-200 p-4 sm:p-6 shadow-xs">
        <div className="max-w-6xl mx-auto space-y-4">
          <div className="flex flex-wrap items-center justify-between gap-4">
            <div className="flex items-center gap-3">
              <div className="p-2.5 bg-amber-50 text-amber-700 rounded-2xl border border-amber-200/80 shadow-xs">
                <HeartHandshake className="w-6 h-6" />
              </div>
              <div>
                <h1 className="text-xl sm:text-2xl font-black text-stone-900 tracking-tight">
                  Quản Lý Hỗ Trợ Gia Đình
                </h1>
                <p className="text-xs text-stone-500 font-medium">
                  Sổ theo dõi quyên góp, trao tặng sinh kế & vật liệu (Đồng bộ thời gian thực)
                </p>
              </div>
            </div>

            {/* Month Select, Backup/Restore & Cloud Sync Status */}
            <div className="flex flex-wrap items-center gap-2 sm:gap-3">
              <select
                className="bg-stone-50 text-stone-800 border border-stone-200 font-mono text-xs font-semibold px-3 py-2 rounded-xl focus:outline-none focus:border-amber-500 cursor-pointer shadow-xs"
                value={month}
                onChange={(e) => setMonth(parseInt(e.target.value, 10))}
              >
                {MONTHS.map((m, i) => (
                  <option key={i} value={i + 1}>
                    {m}
                  </option>
                ))}
              </select>

              <input
                type="file"
                accept=".json,application/json"
                ref={fileInputRef}
                onChange={handleFileSelected}
                className="hidden"
              />
              <button
                type="button"
                onClick={handleBackup}
                title="Sao lưu dữ liệu ra file JSON"
                className="inline-flex items-center gap-1.5 px-3 py-2 rounded-xl bg-stone-50 hover:bg-stone-100 border border-stone-200 text-stone-700 font-semibold text-xs transition-colors shadow-xs"
              >
                <Download className="w-3.5 h-3.5 text-stone-600" />
                <span>Sao Lưu</span>
              </button>
              <button
                type="button"
                onClick={handleRestoreClick}
                title="Khôi phục dữ liệu từ file JSON"
                className="inline-flex items-center gap-1.5 px-3 py-2 rounded-xl bg-stone-50 hover:bg-stone-100 border border-stone-200 text-stone-700 font-semibold text-xs transition-colors shadow-xs"
              >
                <Upload className="w-3.5 h-3.5 text-stone-600" />
                <span>Khôi Phục</span>
              </button>

              <div className="flex items-center gap-2 px-3 py-2 rounded-xl bg-stone-50 border border-stone-200 font-mono text-[11px] shadow-xs">
                <span
                  className={`w-2 h-2 rounded-full ${
                    saveStatus === "saving"
                      ? "bg-amber-500 animate-pulse"
                      : saveStatus === "offline"
                      ? "bg-rose-500"
                      : "bg-emerald-500"
                  }`}
                />
                <span className="text-stone-600 font-medium">
                  {saveStatus === "saving"
                    ? "Đang lưu..."
                    : saveStatus === "offline"
                    ? "Ngoại tuyến"
                    : "Đã đồng bộ"}
                </span>
              </div>
            </div>
          </div>

          {/* Grand Budget Total Summary Banner */}
          <div className="bg-gradient-to-r from-emerald-50 via-emerald-50/80 to-amber-50/50 border border-emerald-200/80 rounded-2xl p-4 flex items-center justify-between shadow-xs">
            <div className="flex items-center gap-3">
              <div className="p-2 bg-emerald-100 text-emerald-700 rounded-xl">
                <Sparkles className="w-5 h-5 animate-pulse" />
              </div>
              <div>
                <span className="text-[10px] font-bold uppercase tracking-wider text-emerald-800 block">
                  Tổng Ngân Sách Dự Chi
                </span>
                <span className="text-xl sm:text-2xl font-extrabold text-emerald-700 font-mono">
                  {formatVnd(grandTotal)}
                </span>
              </div>
            </div>
          </div>
        </div>
      </header>

      {}
      <main className="max-w-6xl mx-auto px-4 sm:px-6 mt-6 space-y-6">
        <div className="flex items-center justify-between gap-4 bg-white border border-stone-200 rounded-2xl p-4 shadow-sm">
          <span className="text-sm font-bold text-stone-700">
            Danh sách các gia đình nhận hỗ trợ
          </span>
          <button
            onClick={addRecord}
            className="inline-flex items-center gap-1.5 px-4 py-2 rounded-xl bg-amber-500 hover:bg-amber-600 text-white font-bold text-xs shadow-md shadow-amber-500/20 transition-all"
          >
            <Plus className="w-4 h-4" /> Thêm Gia Đình
          </button>
        </div>

        <div className="space-y-6">
          {records.map((record, idx) => (
            <RecordCard
              key={record.id}
              record={record}
              index={idx}
              onField={(field, value) => updateField(record.id, field, value)}
              onToggleExpand={() => toggleExpand(record.id)}
              onToggleItem={(itemId, checked) => toggleItem(record.id, itemId, checked)}
              onNameChange={(itemId, name) => changeItemName(record.id, itemId, name)}
              onCostChange={(itemId, cost) => changeItemCost(record.id, itemId, cost)}
              onRequestDeleteItem={(itemId) => requestDeleteItem(record.id, itemId)}
              onAddItem={() => addItem(record.id)}
              onDelete={() => requestDelete(record.id)}
            />
          ))}

          {records.length === 0 && (
            <div className="text-center py-12 bg-white border border-stone-200 rounded-2xl shadow-xs">
              <p className="text-stone-500 text-sm font-medium">
                Chưa có danh sách gia đình nào. Nhấn "+ Thêm Gia Đình" ở trên để bắt đầu.
              </p>
            </div>
          )}
        </div>
      </main>

      <ConfirmModal modal={modal} onConfirm={confirmModal} onCancel={closeModal} />
      <ToastNotification toast={toast} onClose={() => setToast(null)} />
    </div>
  );
}hhh
